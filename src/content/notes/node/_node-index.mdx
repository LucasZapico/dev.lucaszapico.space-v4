# Node

## Custom Error 

```ts
// bad practice
if(!thing){
  throw ("some error")
}

// better 
if(!thing){
  throw new Error("some error")
}

// best pattern 
if (!thing){
  // custom error class
  throw new AppError("some error")
}
```

### Custom App Error 
```ts 
export class AppError extends Error {
  constructor(name: string, httpCode: HttpCode, description: string) {
    
    super(description);
    this.name = name;
    this.httpCode = httpCode;

    Error.captureStackTrace(this);
  })
}
```
